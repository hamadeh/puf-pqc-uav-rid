* RO-PUF HSPICE Monte Carlo simulation
* 45nm CMOS, 32 ROs, two banks of 16, 7-stage inverters

.options post=2 probe
.options monte=100            * 100 device instances
.options seed=12345

* -------- Global parameters --------
.param temp=25                * nominal temperature (°C)
.param vdd=1.0                * nominal supply (V)
.param vth0_nom=0.35          * nominal Vth (NMOS)
.param u0_nom=300e-4          * nominal mobility (cm²/Vs)

* -------- Process variation model (Gaussian) --------
.param vth_std='0.1*vth0_nom' * 10% sigma
.param u0_std='0.1*u0_nom'    * 10% sigma

* -------- Inverter subcircuit (with local variations) --------
.subckt inv in out
  M1 out in 0 0 nmos w=0.2u l=0.05u
  M2 out in vdd vdd pmos w=0.4u l=0.05u
.ends

* -------- 7-stage ring oscillator --------
.subckt ro7 out
  X1 n1 n2 inv
  X2 n2 n3 inv
  X3 n3 n4 inv
  X4 n4 n5 inv
  X5 n5 n6 inv
  X6 n6 n7 inv
  X7 n7 out inv
  * feedback
  rfb out n1 1e6
.ends

* -------- Instantiate 32 ROs (two banks) --------
Xro0  ro0_out ro7
Xro1  ro1_out ro7
...
Xro31 ro31_out ro7

* -------- Multiplexers (simplified as voltage-controlled switches) --------
* (not simulated in detail here; assume ideal mux selection)
* We measure frequency of each RO in separate runs.

* -------- Measurement of frequency --------
.tran 0.1n 100n sweep monte=100

.meas tran freq0 trig v(ro0_out) val=0.5 rise=1 
+              targ v(ro0_out) val=0.5 rise=11
.meas tran freq1 trig v(ro1_out) val=0.5 rise=1 
+              targ v(ro1_out) val=0.5 rise=11
...
* For each pair, compare frequencies.

* -------- PVT analysis --------
.alter temp=-40
.alter temp=100
.alter vdd=0.9
.alter vdd=1.1
* (The above will be swept in separate runs or using .step)

* -------- Model cards (with variation) --------
.model nmos nmos (vth0='vth0_nom+agauss(0,vth_std,1)'
+                u0='u0_nom+agauss(0,u0_std,2)')
.model pmos pmos (vth0='vth0_nom+agauss(0,vth_std,3)'
+                u0='u0_nom+agauss(0,u0_std,4)')

.end
